import express from "express"; import cors from "cors"; import dotenv from "dotenv"; dotenv.config(); const app = express(); app.use(cors()); app.use(express.json()); /* ======================================================== NEXORA AI Sistema de productos + stock + precios + memoria ======================================================== */ // ------------------------------------------------------ // PRODUCTOS // ------------------------------------------------------ const products = [ { id: "phone-x", name: "NEXORA Phone X", price: 899, stock: 25, currency: "EUR", description: "Smartphone premium de NEXORA." }, { id: "phone-pro", name: "NEXORA Phone Pro", price: 1199, stock: 15, currency: "EUR", description: "Smartphone premium avanzado." }, { id: "laptop-pro", name: "Portátil NEXORA Pro", price: 1299, stock: 12, currency: "EUR", description: "Portátil profesional de alto rendimiento." }, { id: "tablet-x", name: "NEXORA Tablet X", price: 699, stock: 20, currency: "EUR", description: "Tablet potente y ligera." } ]; // ------------------------------------------------------ // MEMORIA DE CONVERSACIONES // ------------------------------------------------------ const conversations = new Map(); function getConversation(conversationId) { if (!conversations.has(conversationId)) { conversations.set(conversationId, { messages: [], currentProduct: null }); } return conversations.get(conversationId); } // ------------------------------------------------------ // BUSCAR PRODUCTO // ------------------------------------------------------ function findProduct(text) { if (!text) return null; const normalizedText = text.toLowerCase(); return products.find(product => { const productName = product.name.toLowerCase(); return ( normalizedText.includes(productName) || normalizedText.includes(product.id) ); }); } // ------------------------------------------------------ // DETECTAR SI HABLA DEL PRODUCTO ANTERIOR // ------------------------------------------------------ function refersToPreviousProduct(text) { const normalized = text.toLowerCase(); const references = [ "su", "sus", "ese", "esa", "ese producto", "ese portátil", "ese movil", "ese móvil", "ese teléfono", "el producto", "el portátil", "el movil", "el móvil", "el teléfono", "cuanto cuesta", "cuánto cuesta", "precio", "stock", "unidades" ]; return references.some(reference => normalized.includes(reference) ); } // ------------------------------------------------------ // DETECTAR INTENCIÓN // ------------------------------------------------------ function detectIntent(text) { const normalized = text.toLowerCase(); if ( normalized.includes("precio") || normalized.includes("cuánto cuesta") || normalized.includes("cuanto cuesta") || normalized.includes("coste") || normalized.includes("vale") || normalized.includes("cuánto vale") || normalized.includes("cuanto vale") ) { return "price"; } if ( normalized.includes("stock") || normalized.includes("unidades") || normalized.includes("disponible") || normalized.includes("disponibilidad") || normalized.includes("cuántos quedan") || normalized.includes("cuantos quedan") ) { return "stock"; } if ( normalized.includes("productos") || normalized.includes("qué tenéis") || normalized.includes("que teneis") || normalized.includes("qué tienen") || normalized.includes("catalogo") || normalized.includes("catálogo") ) { return "products"; } return "general"; } // ------------------------------------------------------ // RESPUESTA SOBRE PRODUCTOS // ------------------------------------------------------ function listProducts() { return products .map( product => `• ${product.name} — ${product.price} € — ${product.stock} unidades disponibles` ) .join("\n"); } // ------------------------------------------------------ // PROCESAR MENSAJE // ------------------------------------------------------ function processMessage(message, conversationId) { const conversation = getConversation(conversationId); const productFromMessage = findProduct(message); // ---------------------------------------------- // Si menciona un producto explícitamente // ---------------------------------------------- if (productFromMessage) { conversation.currentProduct = productFromMessage; } // ---------------------------------------------- // Producto actual // ---------------------------------------------- let product = productFromMessage || conversation.currentProduct; const intent = detectIntent(message); // ---------------------------------------------- // STOCK // ---------------------------------------------- if (intent === "stock") { if (!product) { return "Claro. ¿De qué producto quieres consultar el stock?"; } return `El stock de ${product.name} es de ${product.stock} unidades.`; } // ---------------------------------------------- // PRECIO // ---------------------------------------------- if (intent === "price") { if (!product) { return "Claro. ¿De qué producto quieres consultar el precio?"; } return `El precio de ${product.name} es de ${product.price} €.`; } // ---------------------------------------------- // LISTA DE PRODUCTOS // ---------------------------------------------- if (intent === "products") { return `Actualmente tenemos estos productos:\n\n${listProducts()}`; } // ---------------------------------------------- // SALUDOS // ---------------------------------------------- const normalized = message.toLowerCase(); if ( normalized === "hola" || normalized.includes("hola buenas") || normalized.includes("buenas") || normalized.includes("hey") ) { return "¡Hola! Soy NEXORA AI. ¿En qué puedo ayudarte?"; } // ---------------------------------------------- // RESPUESTA GENERAL // ---------------------------------------------- if (product) { return `Claro. Estás consultando sobre ${product.name}. Puedes preguntarme por su precio o stock.`; } return "Puedo ayudarte con nuestros productos, precios y stock. Por ejemplo: «¿Cuál es el precio del NEXORA Phone X?»"; } // ------------------------------------------------------ // API // ------------------------------------------------------ app.post("/chat", (req, res) => { try { const { message, conversationId = "default" } = req.body; if (!message || typeof message !== "string") { return res.status(400).json({ error: "El campo 'message' es obligatorio." }); } const conversation = getConversation(conversationId); // Guardamos el mensaje del usuario conversation.messages.push({ role: "user", content: message }); // Generamos respuesta const response = processMessage( message, conversationId ); // Guardamos respuesta de NEXORA conversation.messages.push({ role: "assistant", content: response }); res.json({ success: true, response, conversationId, currentProduct: conversation.currentProduct ? conversation.currentProduct.name : null }); } catch (error) { console.error("Error:", error); res.status(500).json({ success: false, error: "Ha ocurrido un error interno." }); } }); // ------------------------------------------------------ // OBTENER PRODUCTOS // ------------------------------------------------------ app.get("/products", (req, res) => { res.json({ success: true, products }); }); // ------------------------------------------------------ // OBTENER STOCK // ------------------------------------------------------ app.get("/products/:id/stock", (req, res) => { const product = products.find( p => p.id === req.params.id ); if (!product) { return res.status(404).json({ error: "Producto no encontrado." }); } res.json({ product: product.name, stock: product.stock }); }); // ------------------------------------------------------ // OBTENER PRECIO // ------------------------------------------------------ app.get("/products/:id/price", (req, res) => { const product = products.find( p => p.id === req.params.id ); if (!product) { return res.status(404).json({ error: "Producto no encontrado." }); } res.json({ product: product.name, price: product.price, currency: product.currency }); }); // ------------------------------------------------------ // LIMPIAR CONVERSACIÓN // ------------------------------------------------------ app.delete("/conversation/:id", (req, res) => { conversations.delete(req.params.id); res.json({ success: true, message: "Conversación eliminada." }); }); // ------------------------------------------------------ // SERVIDOR // ------------------------------------------------------ const PORT = process.env.PORT || 3000; app.listen(PORT, () => { console.log(` ======================================== NEXORA AI ONLINE ======================================== Servidor: http://localhost:${PORT} Endpoints: POST /chat GET /products GET /products/:id/stock GET /products/:id/price DELETE /conversation/:id ======================================== `); });
